cmake_minimum_required(VERSION 3.8)
project(odom_visualization)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(Eigen3 REQUIRED)

include_directories(include ${EIGEN3_INCLUDE_DIR})

add_executable(odom_visualization src/odom_visualization.cpp)
ament_target_dependencies(odom_visualization rclcpp sensor_msgs nav_msgs visualization_msgs quadrotor_msgs geometry_msgs)

install(TARGETS odom_visualization DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY meshes DESTINATION share/${PROJECT_NAME})
ament_package()
